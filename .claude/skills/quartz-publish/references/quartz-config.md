## quartz.layout.ts 핵심 설정

```typescript
// 사이드바에 탐색기 추가
Component.Explorer({
  title: "탐색기",
  folderClickBehavior: "collapse",
  folderDefaultState: "collapsed",
  useSavedState: true,
  sortFn: (a, b) => {
    // 폴더 먼저, 그 다음 파일
    if ((!a.file && !b.file) || (a.file && b.file)) {
      return a.displayName.localeCompare(b.displayName, "ko")
    }
    if (a.file && !b.file) return 1
    return -1
  },
})
```
